export interface LaneSnapshot {
  lane: number;
  dots: number;           // 青ドット数（有効な強化効果の種別数）
  instances: number;      // 効果インスタンス数
  danceStages: number;
  danceCap: number;
  danceMultPermil: number;
  effectiveDance: number;
  effects: { type: string; stages: number; endBeat: number; from: string }[];
}

export interface BeatRow {
  beat: number;
  acts: ActivationTrace[];
  comboBefore: number;
  comboAfter: number;
  miss: number;
  lanes: LaneSnapshot[];
  highlight?: string;
  scheduled: ScheduledAct[];
}

export interface ReplayResult {
  rows: BeatRow[];
  maxCombo: number;
  finalCombo: number;
  miss: number;
  checks: { label: string; expected: string; actual: string; ok: boolean }[];
}

function makeState(i: number): LaneState {
  return {
    input: LANES[i],
    effects: [],
    skillCt: new Map(),
    skillUses: new Map(),
    stamina: 1_000_000,
    combo: 0,
    score: 0,
  };
}
